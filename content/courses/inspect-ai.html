    if isinstance(solver, list):
        return chain(solver)
    elif is_agent(solver):
        return as_solver(solver)
    else:
        return cast(Solver, solver)

    def agent_to_solver() -> Solver:
        async def solve(state: TaskState, generate: Generate) -> TaskState:
            agent_state = AgentState(messages=state.messages)

            try:
                # run the agent with limits
                with apply_limits(limits):
                    async with span(name=agent_name, type=AGENT_SPAN_TYPE):
                        agent_state = await agent(agent_state, **agent_kwargs)

            # AgentState's messages + output so that it appears in the log and is scored
            finally:
                # update messages
                state.messages = agent_state.messages

                # update output if its not empty
                if agent_state.output:
                    state.output = agent_state.output

            return state

@metric
def accuracy(to_float: ValueToFloat = value_to_float()) -> Metric:
    r"""Compute proportion of total answers which are correct.

    Args:
       to_float: Function for mapping `Value` to float for computing
          metrics. The default `value_to_float()` maps CORRECT ("C") to 1.0,
          INCORRECT ("I") to 0, PARTIAL ("P") to 0.5, and NOANSWER ("N") to 0,
          casts numeric values to float directly, and prints a warning and returns
          0 if the Value is a complex object (list or dict).

    Returns:
       Accuracy metric
    """

    def metric(scores: list[SampleScore]) -> float:
        if not scores:
            # No scores to average; return 0 rather than dividing by zero,
            # mirroring the insufficient-data guards in std()/var().
            return 0.0
        total = 0.0
        for item in scores:
            total += to_float(item.score.value)
        return total / float(len(scores))

src/inspect_ai/log/_log.py:848-871

白话翻译 total_samples 是计划数；completed_samples 是无错误完成数；取消或 drain 时，logged_samples 还会指出本日志真正解决了多少。只看总数会把中途放弃误读成完整运行。

class EvalResults(BaseModel):
    """Scoring results from evaluation."""

    total_samples: int = Field(default=0)
    """Total samples in eval (dataset samples * epochs)"""

    completed_samples: int = Field(default=0)
    """Samples completed without error.

    Will be equal to total_samples except when --fail-on-error is enabled
    or when there is early stopping.
    """

    logged_samples: int | None = Field(default=None)
    """Samples this log actually resolved (present in the log and not
    cancelled), when a graceful task cancel or drain abandoned queued samples.

    `total_samples` records the *planned* count, so a log finished by
    `inspect ctl task drain` or `inspect ctl task cancel --action score|error`
    would otherwise read complete to an eval set; the eval set's run-vs-reuse
    check prefers this count when present so the abandoned remainder is
    re-run by a later invocation. None on ordinary logs (and logs written by
    older versions), which classify by `total_samples` as before.
    """

Module 1 · Evaluation entry

一条命令，先经过调度台

学完这一节，你能把 inspect eval 从终端入口追到 Python 评测 API；命令失败时，会先判断停在命令注册、参数解析、任务解析，还是样本运行。

本课追踪对象：inspect eval examples/skills/task.py --model openai/gpt-5。它会评测五个 Linux 系统探索问题，而不是启动一个没有判卷规则的自由聊天。

CLI 像火车站调度台：负责认出线路、整理参数并发车，本身不是完成整趟旅程的列车。

The dispatch desk

inspect 先被映射到一个 Python 入口

Single run or eval set

参数整理完，才分流到 eval() 或 eval_set()

shell 根据 pyproject 的 project.scripts 找到 main()。

Applied diagnosis

不要把所有失败都叫“模型坏了”

命令不存在

查安装入口与子命令注册；此时还没有加载 Task。

任务找不到

查路径、注册名与任务发现；此时通常还没有运行样本。

模型无法解析

查 Provider、模型名与认证配置，不要先改 Scorer。

样本或评分失败

此时再进入 Solver、工具、沙箱、Scorer 与日志证据。

Module 2 · Evaluation contract

Task 是实验配方卡

学完这一节，你能把“评测一个 Agent”拆成题目、执行方法、判卷规则和运行环境；新增案例时，会改正确的位置，而不是把预期答案塞进提示词。

一张完整配方卡：Dataset 决定问什么，Solver 决定怎样作答，Scorer 决定怎样判，Sandbox 决定在哪里执行。

Sample 是一份试题；Task 是整套实验配方。

Questions and targets

真实示例先冻结五道题及其目标

五道题共享同一执行和评分合同，因此能比较 Agent 在系统信息、CPU、内存、网络和磁盘问题上的表现。

The complete recipe

同一个 Task 把 Agent、工具、评分与环境扣在一起

Dataset

冻结输入、目标和样本元数据。

Solver

定义 Agent 或求解步骤如何推进状态。

Scorer

把输出与 target 转成结构化 Score。

Sandbox

准备可执行工具的隔离环境和生命周期。

Resolution

运行前，Task 会被解析成不可变的执行单

@task 把工厂函数注册为可发现的评测任务。

Steer the right layer

改实验，先说清你要改变哪一个变量

新增题目

增加 Sample，并写清 input、target 与必要元数据。

改变作答策略

修改 Solver 或 Agent；不要污染 target。

改变判定标准

修改 Scorer、模板或 Metric，并保留旧基线。

改变机器条件

修改 Sandbox 配置，并记录镜像与环境版本。

Try it

要新增“检查 Python 版本”的案例，预期结果最应该放在哪里？

排错提示：样本数不对先查 Dataset 与 limit；行为不对查 Solver；判定不对查 Scorer；命令环境不对查 Sandbox。

Module 3 · Agent as Solver

Agent 先接转接头，再跑循环赛道

学完这一节，你能解释 Agent 如何成为 Inspect 的 Solver、为什么一次回答可能包含多轮模型与工具往返，以及 submit 为什么是明确的终点线。

Solver 是 Inspect 评测管线认识的接口；ReAct Agent 则是一名会反复观察和行动的选手。

关键区别：模型生成了一段文字，不等于样本已提交。这个示例默认要求成功调用 submit 工具才越过终点线。

The adapter

Task 遇到 Agent，会自动把它包成 Solver

The loop

每一圈：收消息、生成、执行工具、把结果放回对话

The finish line

submit 的成功结果才成为待评分答案

这是本样本的 input、target 和初始消息，请 Solver 推进。
我把消息转成 AgentState，并在限制与 trace span 内启动 ReAct。
请结合当前消息与 Skill、Bash、Submit 工具决定下一步。
我执行工具调用，把结果作为 Tool Message 放回对话。
提交成功；答案写入 completion，完整消息与输出交回评分阶段。

Steering and debugging

看它停在哪一圈，而不是只看最后一句

没有工具调用

查提示、可见工具、模型工具能力与 on_continue。

工具调用报错

查参数、审批、沙箱与 Tool Message；不要掩盖错误后继续。

反复循环

查 submit 是否可见、限制是否冻结、失败反馈是否可行动。

异常后日志为空

核对 AgentState 是否在 finally 中回写到 TaskState。

模型写出了正确文字，却一直没有形成可评分提交，最先查什么？

Module 4 · Capability boundaries

隔离柜不等于门卫

学完这一节，你能区分“Agent 能看见哪些工具”“命令在哪里运行”“某次调用是否获准”三件事；设计高风险评测时，不会把 Docker 当成审批策略的替身。

Sandbox 像实验室隔离柜；Approval 像门卫逐单验票。

重要：示例显式配置了 Docker，却没有显式配置 approval policy。这能证明环境选择，不能证明 Bash 命令会等待人工批准。

Capability vs environment

工具列表授予能力，Sandbox 选择落地点

Tool list

定义 Agent 可见的能力表面。

Sandbox

定义能力实际触达的运行环境。

Approval

在具体调用前做逐次策略决策。

Review

对工具结果做后置复核；不能替代执行前审批。

Sandbox lifecycle

每个样本都经历初始化、运行与清理

因此环境启动失败、样本工具失败和环境清理失败是三条不同故障链。排错时应分别保存错误与时间点。

Approval semantics

没有注册审批系统时，调用默认放行

Design the boundary

先写威胁，再选择控制层

怕污染宿主机

冻结 Sandbox 类型、镜像、挂载、网络与清理验证。

怕高风险命令

配置 Approval Policy，覆盖允许、拒绝和未匹配案例。

怕工具结果带毒

使用 Review、输出限制和后续模型输入边界。

怕评测不可复现

记录 Compose、镜像摘要、Skill 内容与外部网络条件。

Try it

目标是允许只读系统查询、拒绝危险 Bash 调用，最直接应改哪一层？

运行边界：静态源码证明这些接缝存在；本课没有验证某个 Compose 文件的实际网络、挂载、Linux capability 或清理效果。

Module 5 · Evidence pipeline

答题、阅卷、统计是三张桌子

学完这一节，你能从 Agent 的 completion 追到单样本 Score，再追到 accuracy 等聚合指标；遇到 grader 协议失败时，会保留“未评分”，而不是伪装成错误答案。

Scorer 是阅卷员；Metric 是统计员。Agent 只负责交卷。

本例：model_graded_qa() 用一个模型判答案，并声明 accuracy 与 stderr 两个指标。

Scorer declaration

评分器在定义时就声明聚合指标与 grader 角色

这让“谁作答”和“谁判卷”可以分离，但也引入新的依赖：grader 模型、模板、正则协议和角色绑定都可能改变结果。

Build the grading prompt

问题、答案与 target 被组装成另一轮模型请求

我根据 Skill 与 Bash 结果提交最终答案。
我把 TaskState.output 与 Sample.target 交给 Scorer。
我按评分模板返回可解析的 C、I、P 或 N。
我保存 value、answer、explanation 和 grading 元数据。
我只聚合有分数的样本，并报告 accuracy 与不确定性。

Protocol failure

读不出 grade，是评分失败，不是答案错误

Aggregation

accuracy 只在样本分数上做透明换算与平均

Try it

grader 返回长篇解释，却没有任何可解析 grade，正确处理是什么？

比较提示：换 grader 模型、模板、grade pattern、reducer 或 Metric，都会改变测量体系；不要把它们和 Agent 能力变化混成一次对比。

Module 6 · Reproducible evidence

Eval Log 是评测黑匣子

学完这一节，你能判断一份日志是否足以支持回归结论：不只看 accuracy，还核对任务、模型、数据集、Scorer、Sandbox、源码 revision、样本状态和错误。

Eval Log 像飞行记录器；回归 不是“这次分数比上次低”这么简单。

可归因比较：先冻结 Task、样本、模型与生成参数、grader、Sandbox、源码 revision，再比较结果与轨迹。

Identity before outcome

日志先记录“这次到底跑了什么”

The black-box layout

身份、计划、结果、统计与错误被分开保存

Planned is not completed

计划样本数与真正完成数必须一起看

EvalSpec 冻结 Task、模型、数据集、Scorer、Sandbox、revision 与包版本。

Regression checklist

先证明可比，再解释差异

身份一致吗？

Task version、源码 revision、包版本、样本 ID 与 epochs。

模型一致吗？

Provider、模型名、角色、生成参数与显式 model args。

测量一致吗？

Scorer、grader、模板、reducer、Metric 与 unscored 处理。

执行一致吗？

Sandbox、Skill、审批、网络、错误、完成数与工具轨迹。

Try it

accuracy 从 0.8 降到 0.6，但模型和源码 revision 都换了，能直接判定框架回归吗？

把这门课变成第一个可运行回归实验

固定这五个 Sample、Inspect commit、Compose/镜像、Agent 与 grader 模型、生成参数和至少三次重复运行；保存完整 Eval Log。升级时只改一个变量，并同时比较 accuracy、stderr、unscored、错误分类、工具调用轨迹、token 与时延。

最终边界：本课验证的是静态调用链与课程本身。真实回归结论仍需要安装 Inspect、可用凭据、Docker、固定外部条件和重复运行证据。
