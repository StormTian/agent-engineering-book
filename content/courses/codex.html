pub async fn compact(sess: &Arc<Session>, sub_id: String) {
    // Stop the old turn before the compact task picks up the next turn's environments.
    sess.abort_all_tasks(TurnAbortReason::Replaced).await;
    let turn_context = sess
        .new_turn_with_default_settings(sub_id, Default::default())
        .await;

    sess.spawn_task(turn_context, Vec::new(), CompactTask).await;
}

/// Runs token-budget manual compaction as a normal compaction lifecycle.
///
/// Token-budget compaction skips model/server summarization and installs a fresh context window
/// instead. It is still modeled as compaction so compact hooks and `ContextCompaction` turn items
/// observe the same lifecycle as local or remote compaction.
pub(crate) async fn run_manual_compact_task(
    sess: Arc<Session>,
    turn_context: Arc<TurnContext>,
) -> CodexResult<()> {
    sess.emit_turn_started(&turn_context).await;

    // Manual compaction runs outside run_turn, so it captures its own current step.
    let step_context = sess
        .capture_step_context(Arc::clone(&turn_context), &CancellationToken::new())
        .await?;
    let world_state = Arc::new(sess.build_world_state_for_step(&step_context).await?);
    run_compact_task_inner(&sess, &step_context, world_state, CompactionTrigger::Manual).await
}

                    continue;
                } else {
                    return Err(e);
                }

fn build_compacted_history_with_limit(
    mut history: Vec<ResponseItemEnvelope>,
    user_messages: &[CompactedUserMessage],
    summary_text: &str,
    max_tokens: usize,
) -> Vec<ResponseItemEnvelope> {
    let mut selected_messages: Vec<CompactedUserMessage> = Vec::new();
    if max_tokens > 0 {
        let mut remaining = max_tokens;
        for message in user_messages.iter().rev() {
            if remaining == 0 {
                break;
            }
            let tokens = approx_token_count(&message.message);
            if tokens <= remaining {
                selected_messages.push(message.clone());
                remaining = remaining.saturating_sub(tokens);
            } else {
                let truncated =
                    truncate_text(&message.message, TruncationPolicy::Tokens(remaining));
                selected_messages.push(CompactedUserMessage {
                    id: message.id.clone(),
                    message: truncated,
                    internal_chat_message_metadata_passthrough: message
                        .internal_chat_message_metadata_passthrough
                        .clone(),
                    harness_metadata: message.harness_metadata.clone(),
                });
                break;
            }
        }
        selected_messages.reverse();
    }

Module 2 · Strategy routing

“压缩”不是一种算法，而是三条分流路径

学完这一节，你能根据 feature 与 provider capability 判断手动或自动压缩会走 token-budget、remote v2 还是 local summarization。

想象搬家：一种直接换到空房间，一种请远端仓库重组，一种在本地整理旧物。三者都叫“搬家”，留下的东西却不相同。

阅读顺序：先判 策略路由，再研究该路径的历史形状；不能拿 local 规则解释所有 compact。

Manual router

手动任务先看 TokenBudget，再问 provider 是否支持 remote v2

Module 4 · Replacement shape

压缩不是缩小每条消息，而是重建一份替代历史

Module 5 · Install and persist

摘要生成成功，仍不等于 session 已经采用它

学完这一节，你能区分生成、安装和持久化三个提交点，并读懂 CompactedItem 如何把 replacement history 变成可恢复的 checkpoint。

Module 6 · Rollout reconstruction

恢复不是从头重放，而是“checkpoint 快照＋存活后缀”

学完这一节，你能解释 resume 如何反向寻找最新可用 checkpoint、何时停止扫描、怎样前向重放后缀，以及旧格式为何需要降级重建。

它像数据库恢复：先找到最近一次完整快照，再只重放快照之后的事务日志；若老快照缺字段，才启用兼容修复。

最终心智模型：rollout reconstruction 的目标不是复刻 UI transcript，而是恢复下一次模型调用所需的语义状态。

Resume entry

InitialHistory::Resumed 会把 rollout 交给统一重建器

Reverse search

先从新到旧，锁定仍然存活的 replacement checkpoint

Snapshot plus suffix

元数据齐全就停止向旧处扫描，再从基线前向重放

从 rollout 尾部向前扫描，按 turn segment 处理回滚。

Legacy fallback

旧 checkpoint 没有 replacement history，只能重新近似构造

调试提示：resume 后上下文位置异常时，先检查 checkpoint 是否缺 replacement_history；不要先把问题归因于“摘要模型忘了”。

Applied check

把静态理解变成下一轮可运行实验

最小 runtime 实验应固定一组早期工具输出、近期用户约束与压缩后新增消息：执行 compact，保存 rollout，关闭并 resume，再分别核对 live history、checkpoint replacement 与 suffix。这样才能测出“生成成功但未持久化”或“恢复选错基线”等差异。
